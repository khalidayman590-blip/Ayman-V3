# فَصْلُ الشَّجَاعَةِ وَالقُوَّة

**الباب الخامس: فَلْسَفَةُ الشَّجَاعَةِ وَكَبَوَاتُ الأَخُوةِ** — القصيدة الأولى
**الموضوع:** الفصل الأخلاقي والفلسفي بين القوة والشجاعة: القوة قدرة قد يملكها البطش، أمّا الشجاعة فحكمة وضبط للنفس ونصرة للضعيف وثبات عند الحق. لا يدخلها عتاب الأخوة؛ له القصيدة الثانية.
**البحر:** المسحوب (مستفعلن مستفعلن فاعلاتن)
**القافية:** الصدر (ـارْ) في كل بيت / العجز (ـودْ) في كل بيت
**الضمير:** موحّد — الغائب العام (هو / من / الشجاع) من أول بيت لآخر بيت
**عدد الأبيات:** ٢٢

---

**١**
ما كلّ من شدّ العزومِه جَسّار
وْلا كلّ صَوْلة تِستحقّ الخلود

**٢**
البَطْش بليا حكمة يِشْعل بها النار
وِالشَّهْم يِطْفيها بِحِلْمٍ وْوجود

**٣**
كم واحدٍ يِبْطش وْيِحْسب له انتصار
وْهو مِن هواجيسِه ضعيفِ القيود

**٤**
وْكم ساكتٍ لا جا نهارِه ولا حار
وِقْفِة جبل ما هزّ راسِه رعود

**٥**
الشُّجاع ما يِظلم مساكين وْلا جار
وْلا يِرْفع إيدِه لِمَنْ هو مَصدود

**٦**
يِعْطي الأمان لِمَنْ تعثّر وْلا ثار
وْيِستر عيوبِ الخلق سترِ السدود

**٧**
وْيِردّ غيظِه لو تمادى به المرار
وْيِحْكم ضميره قبل حكمِ الشهود

**٨**
وِالشِّدّة اللي ما لها رُشْد مِسمار
تِنْهدّ لو بانِت صلابِة حشود

**٩**
وِالجَسور راعي موقفٍ وقت الأخطار
يِقْدم وْلا يِسأل عن كَثْرة عُدود

**١٠**
ما يِطْلب المدحة وْلا يِرْخص أسرار
وْلا يِشْتري صيتِه بِزيف الوعود

**١١**
يِمشي طريقِ المراجل بِمقدار
وْيِصون حدّ الحق عن كلّ زود

**١٢**
إن قال كلمة ما تِبدّل بالأعذار
وْإن عاهدِ الوافي وفى بِالعهود

**١٣**
ما هو كفوٍّ مَنْ يِذِلّ له صغار
وْلا هو قويٍّ مَنْ بِبيته يِسود

**١٤**
هي نخوة وْفِطْنة وْتَدْبير وِأفكار
وْنَجْدة ملهوفٍ تِعَنّى وْمكدود

**١٥**
وِالنِّشْمي يِعْرف متى يصبر وِيختار
وْوين يِقول الحق بِوجه العنود

**١٦**
ما هاب قول الناس لو صار محتار
دامِ الدليلِ البيّنِ له موجود

**١٧**
وْلا ساقه الغضب العمى صوبِ الأشرار
وْلا طاع نَفْسٍ للمذلّة تِقود

**١٨**
يِعْفي وهو قادِر وْلا شاف له عار
لأن العفو عند النشامى مَحمود

**١٩**
لكن ليا جا للكرامة يدٍ ضار
يِذود عنها وْالسلايل تِذود

**٢٠**
هذي الشجاعة ما هي بطشٍ وْإجبار
هي ثِقْل رأيٍ في نهارٍ مَجهود

**٢١**
مَنْ ضبَط نَفْسِه عن خطاها صَبّار
وْمَنْ نَصَر المظلوم فعله مَشهود

**٢٢**
وِالختم ذِكر الله في سرّ وِإجهار
وْصلاة ربّي عالنبيّ خيرِ مولود
