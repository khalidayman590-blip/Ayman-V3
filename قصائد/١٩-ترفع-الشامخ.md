# تَرُفُّع الشَّامِخ

**الباب الخامس: فَلْسَفَةُ الشَّجَاعَةِ وَكَبَوَاتُ الأَخُوةِ** — القصيدة الثالثة
**الموضوع:** ترفّع الشامخ عن ردّ الإساءة بالإساءة: يرى الخذلان ويقدر على الرد، لكنه يختار الصمت والحلم وصون المقام وحماية لمّة الأهل. لا يعيد عتاب الأخوة المباشر من القصيدة السابقة.
**البحر:** المسحوب (مستفعلن مستفعلن فاعلاتن)
**القافية:** الصدر (ـيبْ) في كل بيت / العجز (ـانْ) في كل بيت
**الضمير:** موحّد — الغائب العام (الشامخ / هو / يـ) من أول بيت لآخر بيت
**عدد الأبيات:** ٢٢

---

**١**
الشامخ اللي ما تِزعزعْه صَعيب
وْلا يِنحني لو جار وقتٍ وْخان

**٢**
يِشوف فعل الردي وْيِعرف المَعيب
وْيِمرّه علوٍّ عن دروب الهوان

**٣**
ما هو عجز لا صار ردّه يِغيب
لكن مقامه ما يِطاوعه يِهان

**٤**
يِقْدر يِصدّ الطعن جرحٍ وْتَعذيب
وِيِعْفي لأن العفو سِمّة شجعان

**٥**
وِيِصون سِرّ البيت لو هو غريب
وْيِسْدل لِزلّات ربعه كِتْمان

**٦**
ما شال في صدره لِلخلق حسيب
وْلا خَلّى البغضا تِقوده لِعدوان

**٧**
يِوزن حكيه وْما يِصيرِه رقيب
وْيِردّ غيظه لا تحرّك به الشان

**٨**
وْإن شاف مِن بعض القرابة عجيب
يِحفظ حبال الودّ مِن حدّ سِكّان

**٩**
وِيِعطي لِمَنْ عاده لِطيبِه قريب
وْلا يِمْنع المعروف مِن خوف خذلان

**١٠**
رأيِه بِدرب المراجل صَليب
وْحِلْمِه مع عثرة رفيقه له برهان

**١١**
ما كلّ طعنة تِستحقّ النحيب
وْلا كلّ زلّة تِشْعل القلب نيران

**١٢**
بعض السكوت لِصاحب العقل طَبيب
وْبعض الحكي يِهدّ المودة وِالأركان

**١٣**
وِالحرّ يِميّز مَنْ خَسيسٍ وْنَجيب
وْيِحطّ كِلٍّ في قدر صُنْعه ميزان

**١٤**
ما هو يِبي كسب المديح مستجيب
وْلا ينتظر مِن كلّ جاحد له إحسان

**١٥**
يِخْطي ولكن ما يِصِرّ وِيستريب
وْإن بان له وجه الخطا قال غلطان

**١٦**
وِيِمُدّ كفّه لِلمخاصم وِالحبيب
وْيِطفي لهيب الشّر قبلِه بِنسيان

**١٧**
وْلا يِحْسب طيب الجميل عَطيب
وْلا يِندم إن جاد لِلضّيف جودان

**١٨**
وِالشَّهم ما يِرضى يعيشِه كئيب
وْيِفتح لِوجه الصبح بابٍ وْرضوان

**١٩**
لو صاح صوت الحقد حوله رَهيب
يِبقى بِنهج النشامى ولا لان

**٢٠**
راسه فَوْق روس النوايف مَهيب
وْفِعله لِعزّ أهله حصنٍ وْبُنيان

**٢١**
هذا سُموّ الحرّ وْفِعله أديب
وْلا هو هروبٍ مِن حقوقٍ وْدَيان

**٢٢**
وِالختم ذِكر الله مولاي وِالمُجيب
وْصلاة ربّي عالنبيّ ولد عدنان
